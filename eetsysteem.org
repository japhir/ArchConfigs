* Reading lists
A list is a level-1 heading; its level-2 children are sections (stores,
Menu). Only checkbox lines count as items.
#+begin_src emacs-lisp
  (defconst eet--checkbox-re
    "^[ \t]*[-+] \\[\\([ X-]\\)\\][ \t]*\\(.*?\\)[ \t]*$"
    "A checkbox item; group 1 is the box state, group 2 the text.")

  (defun eet--sections ()
    "Checkbox items under the heading at point, grouped by child heading.
  Return an ordered alist (TITLE . ITEMS); each item is (DONE . TEXT).
  Empty items are dropped."
    (save-excursion
      (org-back-to-heading t)
      (let ((child (1+ (org-outline-level)))
            (end (save-excursion (org-end-of-subtree t t)))
            acc)
        (forward-line)
        (while (< (point) end)
          (cond
           ((org-at-heading-p)
            (when (= (org-outline-level) child)
              (push (list (org-get-heading t t t t)) acc)))
           ((and acc (looking-at eet--checkbox-re))
            (let ((text (match-string-no-properties 2)))
              (unless (string-empty-p text)
                (push (cons (equal (match-string 1) "X") text) (cdar acc))))))
          (forward-line))
        (nreverse (mapcar (lambda (s) (cons (car s) (reverse (cdr s)))) acc)))))

  (defun eet--title ()
    "Heading text at point, without keyword, priority, tags, comment or
  text properties (fontification would travel along into inserted links)."
    (substring-no-properties (org-get-heading t t t t)))

  (defun eet--find-level-1 (pred)
    "Markers of level-1 headings in the buffer satisfying PRED, in order."
    (let (found)
      (org-with-wide-buffer
       (goto-char (point-min))
       (while (re-search-forward "^\\* " nil t)
         (beginning-of-line)
         (when (funcall pred) (push (point-marker) found))
         (end-of-line)))
      (nreverse found)))

  (defun eet--open-lists ()
    "Markers of the shopping lists that are not done yet."
    (eet--find-level-1
     (lambda () (and (string-prefix-p eet-list-prefix (eet--title))
                     (not (org-entry-is-done-p))))))

  (defun eet--staples-heading ()
    "Marker of the Basisvoorraad heading, or nil."
    (car (eet--find-level-1
          (lambda () (equal (eet--title) eet-staples-heading)))))

  (defun eet--staples ()
    "Alist (STORE . ITEMS) from the Basisvoorraad heading.
  Each item is (DONE . TEXT); a tick means we still have it."
    (when-let* ((m (eet--staples-heading)))
      (org-with-point-at m
        (eet--sections))))
#+end_src

* Starting a new week
Before planning we walk through the house (the Huisronde) and tick the
staples we still have. Only unticked staples go onto the new list, and a
ticked staple also stops an unbought item of the same name from carrying
over: the walk is newer than last week's list. Afterwards every staple is
unticked again, ready for next week's walk.

Unbought items move to the same store in the new list, or to the first
store from a section that is no store. Carried items come
first, then the staples that aren't already there. Every checkbox section
ends with one empty =- [ ]=: on beorg and orgzly, return on a checkbox line
makes the next one, while typing =- [ ]= from nothing on a phone keyboard
is a pain.
#+begin_src emacs-lisp
  (defun eet--norm (s)
    "S lowercased without whitespace, for matching names and items."
    (downcase (replace-regexp-in-string "[[:space:]]+" "" s)))

  (defun eet--stores (staples)
    "`eet-stores' plus any extra store in STAPLES."
    (seq-uniq (append eet-stores (mapcar #'car staples))))

  (defun eet--carried (lists stores)
    "Alist (SECTION . TEXTS) of the unticked items in LISTS (markers)."
    (let (acc)
      (dolist (m lists)
        (org-with-point-at m
          (pcase-dolist (`(,sec . ,items) (eet--sections))
            (let ((target (cond ((equal sec eet-menu-section) sec)
                                ((equal sec eet-leftovers-section) nil)
                                (t (or (car (member sec stores)) (car stores))))))
              (when target
                (dolist (it items)
                  (unless (car it)
                    (push (cdr it) (alist-get target acc nil nil #'equal)))))))))
      (mapcar (lambda (c) (cons (car c) (reverse (cdr c)))) acc)))

  (defun eet--merge (&rest lists)
    "Concatenate LISTS of item texts, dropping repeats by `eet--norm'."
    (let (seen out)
      (dolist (it (apply #'append lists) (nreverse out))
        (unless (member (eet--norm it) seen)
          (push (eet--norm it) seen)
          (push it out)))))

  (defun eet--render (title time sections)
    "Org text of a list TITLE created at TIME.
  SECTIONS is a list of (NAME CHECKBOXES . ITEMS)."
    (concat
     "* TODO " title "\n"
     ":PROPERTIES:\n:CREATED:  "
     (format-time-string "[%Y-%m-%d %a %H:%M]" time) "\n:END:\n"
     (mapconcat
      (pcase-lambda (`(,name ,boxes . ,items))
        (concat "** " name "\n"
                (mapconcat (lambda (it) (concat (if boxes "- [ ] " "- ") it "\n"))
                           items "")
                (if boxes "- [ ]\n" "")))
      sections "")))

  (defun eet--week-title (time)
    "Shopping list title for a list started at TIME.
  Named by date, not ISO week: planning on Saturday still falls in the week
  of the previous list."
    (concat eet-list-prefix (format-time-string "%Y-%m-%d" time)))

  (defun eet-new-week (&optional time)
    "Start this week's shopping list in eetplanning.org and show it.
  Close the Menu of every open list (see `eet--close-menus'), carry their
  unticked items over, add the unticked
  staples from Basisvoorraad, then mark the old lists done, archive them
  and untick every staple. TIME defaults to now."
    (interactive)
    (let* ((buf (find-file-noselect (eet-planning-file)))
           (pos (with-current-buffer buf (eet--new-week time))))
      (when (called-interactively-p 'interactive)
        (pop-to-buffer-same-window buf)
        (goto-char pos))))

  (defun eet--new-week (time)
    "`eet-new-week' in the current buffer.
  Return the position of the new list's Nog in huis items."
    (let* ((time (or time (current-time)))
           (title (eet--week-title time))
           (old (eet--open-lists)))
      (when (seq-some (lambda (m) (org-with-point-at m (equal (eet--title) title)))
                      old)
        (user-error "%s already exists" title))
      (let* ((cooked (eet--close-menus old))
             (staples (eet--staples))
             (stores (eet--stores staples))
             (carried (eet--carried old stores))
             (get (lambda (al k) (alist-get k al nil nil #'equal)))
             (have (mapcar (lambda (it) (eet--norm (cdr it)))
                           (seq-filter #'car (apply #'append (mapcar #'cdr staples)))))
             (keep (lambda (texts)
                     (seq-remove (lambda (it) (member (eet--norm it) have)) texts)))
             (sections
              `((,eet-leftovers-section nil)
                (,eet-menu-section t ,@(funcall get carried eet-menu-section))
                ,@(mapcar (lambda (st)
                            `(,st t ,@(funcall keep
                                               (eet--merge
                                                (funcall get carried st)
                                                (mapcar #'cdr (funcall get staples st))))))
                          stores)))
             new)
        (org-with-wide-buffer
         (goto-char (point-max))
         (unless (bolp) (insert "\n"))
         (setq new (point-marker))
         (insert (eet--render title time sections)))
        (let ((org-enforce-todo-dependencies nil)
              (org-enforce-todo-checkbox-dependencies nil))
          ;; in buffer order: each archive is appended, and
          ;; `eet--last-winkels' takes the last one as newest
          (dolist (m old)
            (org-with-point-at m
              (org-todo 'done)
              (org-archive-subtree))))
        (when-let* ((m (eet--staples-heading)))
          (org-with-point-at m (org-reset-checkbox-state-subtree)))
        (eet--log-cooked cooked time)
        (widen)
        (goto-char new)
        (re-search-forward (concat "^\\*\\* " eet-leftovers-section "\n") nil t)
        (org-fold-show-subtree)
        (point))))
#+end_src

* Naampjes
A recipe's =ID= is its naampje (a slug): a short name we can type from
memory in the kitchen: =- [ ] pompoencurry= on the Menu, from a phone, which the weekly
close turns into an =id:= link. org-id accepts any string, so links work
across files just like with uuids. Lowercase ASCII, hyphens only between
words. The suggestion is the first word of the title without accents;
titles are long, so it is meant to be edited. Template R leaves =:ID:=
empty and calls this before finishing; recipes added on a phone with just a
name get theirs from =M-x eet-set-naampje=. The new naampje goes straight
into org-id's table: following a link to an ID it doesn't know makes
org-id rescan every file.
#+begin_src emacs-lisp
  (require 'ucs-normalize)
  (require 'org-id)

  (defconst eet--naampje-re "\\`[a-z0-9]+\\(?:-[a-z0-9]+\\)*\\'"
    "Shape of a naampje; `eet--naampje-p' also rules out uuids.")

  (defconst eet--uuid-re
    "\\`[0-9a-f]\\{8\\}\\(?:-[0-9a-f]\\{4\\}\\)\\{3\\}-[0-9a-f]\\{12\\}\\'"
    "An org-id uuid, which also has the shape of a naampje.")

  (defun eet--naampje-p (id)
    "Whether ID is a naampje, not a uuid."
    (let ((case-fold-search nil))
      (and (string-match-p eet--naampje-re id)
           (not (string-match-p eet--uuid-re id)))))

  (defun eet--plain (s)
    "S lowercase, without accents, keeping only letters, digits, spaces, hyphens."
    (replace-regexp-in-string
     "[^a-z0-9 -]" "" (downcase (ucs-normalize-NFD-string s))))

  (defun eet--suggest-naampje (title)
    "Suggested naampje for TITLE: its first word, lowercase, without accents."
    (let ((word (car (split-string (eet--plain title) nil t))))
      (string-trim (or word "") "-+" "-+")))

  (defun eet--recipes-here ()
    "Alist (ID . TITLE) of the level-1 headings with an ID in this buffer."
    (seq-keep (lambda (m)
                (org-with-point-at m
                  (let ((id (org-entry-get nil "ID")))
                    (and id (not (string-empty-p id)) (cons id (eet--title))))))
              (eet--find-level-1 #'always)))

  (defun eet--recipe-ids ()
    "IDs of the level-1 headings in the current buffer."
    (mapcar #'car (eet--recipes-here)))

  (defun eet--naampje-problem (naampje taken)
    "Why NAAMPJE can't be used given TAKEN ids, or nil when it can."
    (cond ((not (eet--naampje-p naampje))
           "lowercase letters, digits and single hyphens only")
          ((member naampje taken) "already used by another recipe")))

  (defun eet-set-naampje ()
    "Ask for a naampje and store it as the ID of the recipe at point.
  Leave a recipe that already has an ID alone."
    (interactive)
    (save-excursion
      (org-back-to-heading t)
      (while (org-up-heading-safe))
      (let ((id (org-entry-get nil "ID")))
        (when (or (null id) (string-empty-p id))
          (let* ((taken (eet--recipe-ids))
                 (naampje (read-string "Naampje: " (eet--suggest-naampje (eet--title))))
                 problem)
            (while (setq problem (eet--naampje-problem naampje taken))
              (setq naampje (read-string (format "Naampje (%s): " problem) naampje)))
            (org-entry-put nil "ID" naampje)
            ;; else following a link to it makes org-id rescan every file
            (when-let* ((file (buffer-file-name (buffer-base-buffer))))
              (org-id-add-location naampje file)))))))
#+end_src

* The Menu
The Menu links recipes with =id:= links, so the cooking log lands on the
recipe itself. =M-x eet-add-to-menu= picks a recipe and puts it on this
week's Menu, above the empty checkbox. Recipes without a naampje can't be
linked, so they aren't offered.

Recipes are looked up in =recepten.org= only, never through org-id's
global table: a missing ID there makes org-id rescan every agenda file.
#+begin_src emacs-lisp
  (defun eet--existing-buffer (file)
    "Buffer visiting FILE, or nil when FILE doesn't exist."
    (and (file-exists-p file) (find-file-noselect file)))

  (defun eet--recipes-buffer ()
    "Buffer visiting recepten.org, or nil when there is none."
    (eet--existing-buffer (eet-recipes-file)))

  (defun eet--recipes ()
    "Alist (ID . TITLE) of the recipes in recepten.org with a naampje.
  Other IDs (uuids) can't be typed on a phone, so they are left out."
    (when-let* ((buf (eet--recipes-buffer)))
      (seq-filter (lambda (r) (eet--naampje-p (car r)))
                  (with-current-buffer buf (eet--recipes-here)))))

  (defun eet--link (recipe)
    "Org id link to RECIPE, an (ID . TITLE) pair."
    (format "[[id:%s][%s]]" (car recipe) (cdr recipe)))

  (defun eet--choices (recipes)
    "Alist (LABEL . RECIPE) for picking one of RECIPES by name or naampje."
    (mapcar (lambda (r) (cons (format "%s (%s)" (cdr r) (car r)) r)) recipes))

  (defun eet--goto-section (name)
    "Move to the child heading NAME of the heading at point, or signal a
  user error."
    (let ((child (1+ (org-outline-level)))
          (parent (eet--title))
          (end (save-excursion (org-end-of-subtree t t)))
          found)
      (while (and (not found) (outline-next-heading) (< (point) end))
        (setq found (and (= (org-outline-level) child)
                         (equal (eet--title) name))))
      (unless found (user-error "No %s section under %s" name parent))))

  (defun eet--insert-item (text)
    "Add checkbox TEXT to the section at point, above its empty checkbox.
  A section without one gets one, so it stays last."
    (let ((item (concat "- [ ] " text "\n")))
      (goto-char (org-entry-end-position))
      (skip-chars-backward " \t\n")
      (forward-line 0)
      (if (looking-at "[ \t]*[-+] \\[ \\][ \t]*$")
          (insert item)
        (forward-line 1)
        (unless (bolp) (insert "\n"))
        (insert item "- [ ]\n"))))

  (defun eet--this-week ()
    "Marker of this week's list, the last open one, or signal a user error."
    (or (car (last (eet--open-lists)))
        (user-error "No open shopping list; run eet-new-week")))

  (defun eet-add-to-menu ()
    "Pick a recipe and add it to the Menu of this week's list."
    (interactive)
    (let ((choices (eet--choices (eet--recipes))))
      (unless choices
        (user-error "No recipes with a naampje in %s" (eet-recipes-file)))
      (with-current-buffer (find-file-noselect (eet-planning-file))
        (let ((list (eet--this-week))
              (recipe (cdr (assoc (completing-read "Recipe: " choices nil t)
                                  choices))))
          ;; RET on empty input passes require-match
          (unless recipe (user-error "No recipe picked"))
          (org-with-point-at list
            (eet--goto-section eet-menu-section)
            (eet--insert-item (eet--link recipe)))
          (message "%s is on the %s" (cdr recipe) eet-menu-section)))))
#+end_src

* Recipe to shopping list
=M-x eet-add-ingredients= with point in a recipe, or on an =id:= link to
one (a Menu line), puts its Benodigdheden on this week's list, each above the empty checkbox of its Winkel, as written:
the quantity helps in the shop. The Winkel is where the item last sat:
Basisvoorraad first, since it is kept up to date, then the archived lists,
newest first. An item we never bought goes to the first store (Lidl/Hira).
Items are compared by name: without the leading quantity and unit, the
notes in brackets and anything after a comma, so =2 uien, gesnipperd=
matches =uien=. Every ingredient is added, even when the list already has
it: =3 uien= and =1 ui= side by side keep both quantities, where adding
them up would trip over units and plurals.

Nothing changes before every item has a section to go to, and
=eetplanning.org= isn't saved, just like =eet-add-to-menu=.
#+begin_src emacs-lisp
  (defconst eet-ingredients-section "Benodigdheden"
    "Recipe section with the ingredients, one plain list item each.")

  (defcustom eet-units
    '("g" "gr" "gram" "kg" "ml" "cl" "dl" "l" "liter" "el" "tl" "eetlepel"
      "eetlepels" "theelepel" "theelepels" "cm" "blik" "blikken" "blikje"
      "blikjes" "pot" "potje" "potjes" "pak" "pakje" "pakjes" "zak" "zakje"
      "zakjes" "bos" "bosje" "bosjes" "teen" "teentje" "teentjes" "tenen"
      "stuk" "stuks" "snuf" "snufje" "plak" "plakken" "takje" "takjes"
      "handje" "handvol" "scheut" "scheutje")
    "Units that may follow the leading number of an ingredient."
    :type '(repeat string))

  (defconst eet--list-item-re
    "^[ \t]*[-+][ \t]+\\(?:\\[[ X-]\\]\\)?[ \t]*\\(.*[^ \t\n]\\)?[ \t]*$"
    "A plain list item, box or not; group 1 is the text, nil when empty.")

  (defun eet--item-name (text)
    "Name of item TEXT for matching: without the leading quantity and unit,
  bracketed notes and anything after a comma, then `eet--norm'."
    (let* ((qty (concat "\\`[[:space:]]*[0-9½¼¾⅓⅔][0-9½¼¾⅓⅔.,/–-]*"
                        "\\(?:[[:space:]]*" (regexp-opt eet-units) "\\>\\)?"))
           (s (replace-regexp-in-string qty "" (downcase text)))
           (s (replace-regexp-in-string "([^)]*)" "" s)))
      (eet--norm (car (split-string s ",")))))

  (defun eet--ingredients ()
    "(TITLE . ITEMS) of the recipe at point: its title and ingredient texts."
    (save-excursion
      (org-back-to-heading t)
      (while (org-up-heading-safe))
      (let ((title (eet--title))
            acc)
        (eet--goto-section eet-ingredients-section)
        (let ((end (save-excursion (org-end-of-subtree t t))))
          (while (< (point) end)
            (when (and (looking-at eet--list-item-re) (match-string 1))
              (push (match-string-no-properties 1) acc))
            (forward-line)))
        (cons title (nreverse acc)))))

  (defun eet--link-id-at-point ()
    "The ID an id: link at point points to, or nil."
    (let ((link (org-element-context)))
      (and (eq (org-element-type link) 'link)
           (equal (org-element-property :type link) "id")
           (org-element-property :path link))))

  (defun eet--recipe-ingredients (id)
    "`eet--ingredients' of recipe ID in recepten.org, or signal a user error."
    (or (when-let* ((buf (eet--recipes-buffer)))
          (with-current-buffer buf
            (org-with-wide-buffer
             (when-let* ((pos (org-find-property "ID" id)))
               (goto-char pos)
               (eet--ingredients)))))
        (user-error "No recipe %s in %s" id (eet-recipes-file))))

  (defun eet--seen-in (markers stores)
    "(NAME . STORE) for each item under MARKERS (lists or Basisvoorraad).
  NAME is an `eet--item-name'; sections that are no store, like Menu, are
  skipped."
    (let (acc)
      (dolist (m markers (nreverse acc))
        (org-with-point-at m
          (pcase-dolist (`(,sec . ,items) (eet--sections))
            (when (member sec stores)
              (dolist (it items)
                (push (cons (eet--item-name (cdr it)) sec) acc))))))))

  (defun eet--last-winkels (list stores)
    "Alist (NAME . STORE) of where items last sat, the most recent first.
  LIST is this week's list; its archive file holds the older ones."
    (let ((archive (org-with-point-at list
                     ;; internal, but it is what org-archive-subtree uses
                     (car (org-archive--compute-location
                           (or (org-entry-get nil "ARCHIVE" 'inherit)
                               org-archive-location))))))
      (append
       (eet--seen-in (ensure-list (eet--staples-heading)) stores)
       (when-let* ((buf (eet--existing-buffer archive)))
         (with-current-buffer buf
           ;; no auto-mode for .org_archive; org-archive does the same
           (unless (derived-mode-p 'org-mode) (org-mode))
           (eet--seen-in
            (reverse (eet--find-level-1
                      (lambda () (string-prefix-p eet-list-prefix (eet--title)))))
            stores))))))

  (defun eet-add-ingredients ()
    "Put the ingredients of the recipe at point, or linked at point, on
  this week's list."
    (interactive)
    (pcase-let ((`(,title . ,items)
                 (if-let* ((id (eet--link-id-at-point)))
                     (eet--recipe-ingredients id)
                   (eet--ingredients))))
      (unless items (user-error "No ingredients under %s" eet-ingredients-section))
      (with-current-buffer (find-file-noselect (eet-planning-file))
        (let* ((list (eet--this-week))
               (stores (eet--stores (eet--staples)))
               (seen (eet--last-winkels list stores))
               (list-title (org-with-point-at list (eet--title)))
               (sections (org-with-point-at list (eet--sections)))
               (plan (mapcar (lambda (it)
                               (cons (or (cdr (assoc (eet--item-name it) seen))
                                         (car stores))
                                     it))
                             items)))
          (dolist (p plan)
            (unless (assoc (car p) sections)
              (user-error "No %s section in %s" (car p) list-title)))
          (dolist (p plan)
            (org-with-point-at list
              (eet--goto-section (car p))
              (eet--insert-item (cdr p))))
          (message "%d ingredients of %s on the list" (length plan) title)))))
#+end_src

* Closing the week
Starting a new week closes the Menu of the old lists first; there is no
separate close command, because closing twice would log every Kookbeurt
twice, and new-week archives the list right after, so each list is closed
exactly once. Per Menu item:

- A ticked item that is plain text, typed on a phone, becomes an =id:=
  link. A naampje (any case: phone keyboards capitalise) or a full recipe
  name is a sure match; otherwise the recipes whose name holds all the
  typed words are offered, or every recipe when none does (a typo), with
  the option to leave the text as it is.
- A ticked recipe asks for ★1–5 and a verdict, and gets a LOGBOOK note
  =Gekookt ★n. verdict= in =recepten.org=, stamped with the planning time
  (the cooking day isn't known, so derived "last cooked" dates run up to a
  week late). The period is always there, even without a verdict, so notes
  can be added after it by hand; unlike a dash it is easy to type. Ticked
  items left as text are listed in a message, since their Kookbeurt isn't
  logged.
- An unticked item asks whether to carry it to next week; if not, the line
  is removed so it isn't carried. A carried plain item is matched like a
  ticked one, except that text matching no recipe name (=pizza bij Jan=)
  stays text without asking.

All questions come first and only then is anything changed, so =C-g=
halfway leaves both files as they were. The notes are written last, after
archiving, so a failure halfway never logs a Kookbeurt that a rerun logs
again. =recepten.org= is saved when it had no unsaved changes before: the
phones should see the log, and saving the user's own half-done edits is
not ours to do.
#+begin_src emacs-lisp
  (defconst eet--leave-as-text "(leave as text)"
    "Choice that keeps a Menu item as plain text.")

  (defun eet--menu-items ()
    "(MARKER DONE TEXT) for each checkbox with text in the list at point's Menu."
    (save-excursion
      (let ((child (1+ (org-outline-level)))
            (end (save-excursion (org-end-of-subtree t t)))
            in acc)
        (forward-line)
        (while (< (point) end)
          (cond ((org-at-heading-p)
                 (when (= (org-outline-level) child)
                   (setq in (equal (eet--title) eet-menu-section))))
                ((not in))
                ((and (looking-at eet--checkbox-re)
                      (not (string-empty-p (match-string 2))))
                 (push (list (point-marker) (equal (match-string 1) "X")
                             (match-string-no-properties 2))
                       acc)))
          (forward-line))
        (nreverse acc))))

  (defun eet--resolve (text recipes cooked)
    "The recipe in RECIPES that Menu item TEXT names, or nil. May ask.
  When no recipe name holds all of TEXT's words, ask among all RECIPES if
  the item was COOKED, or else give up."
    (let ((n (eet--norm text))
          (words (split-string (eet--plain text) "[ -]+" t)))
      (or (seq-find (lambda (r) (equal (eet--norm (car r)) n)) recipes)
          (seq-find (lambda (r) (equal (eet--norm (cdr r)) n)) recipes)
          (when-let* ((choices
                       (eet--choices
                        (or (seq-filter
                             (lambda (r)
                               (let ((hay (eet--plain (concat (cdr r) " " (car r)))))
                                 (seq-every-p (lambda (w) (string-search w hay)) words)))
                             recipes)
                            (and cooked recipes)))))
            (cdr (assoc (completing-read (format "Which recipe is \"%s\"? " text)
                                         (cons eet--leave-as-text choices) nil t)
                        choices))))))

  (defun eet--ask-verdict (name)
    "Ask for stars and a verdict on NAME; return the cooking note."
    (let ((stars "")
          verdict)
      (while (not (member stars '("1" "2" "3" "4" "5")))
        (setq stars (completing-read (format "★ for %s (1-5): " name)
                                     '("1" "2" "3" "4" "5") nil t)))
      (setq verdict (string-trim (read-string (format "Verdict on %s: " name))))
      (concat "Gekookt ★" stars "."
              (if (string-empty-p verdict) "" (concat " " verdict)))))

  (defun eet--close-menus (lists)
    "Close the Menu of each list in LISTS (markers): link, carry or drop.
  Return the cooking notes to log, as (ID . NOTE) pairs."
    (let ((recipes (eet--recipes))
          plan cooked unlogged)
      (dolist (m lists)
        (org-with-point-at m
          (pcase-dolist (`(,pos ,done ,text) (eet--menu-items))
            (let* ((linked (and (string-match "\\[\\[id:\\([^]]+\\)\\]" text)
                                (match-string 1 text)))
                   (drop (and (not done)
                              (not (y-or-n-p (format "Carry %s over to next week? "
                                                     text)))))
                   (recipe (cond (drop nil)
                                 (linked (assoc linked recipes))
                                 (t (eet--resolve text recipes done)))))
              (push (list pos drop (and recipe (not linked) (eet--link recipe)))
                    plan)
              (when done
                (if recipe
                    (push (cons (car recipe) (eet--ask-verdict (cdr recipe))) cooked)
                  (push text unlogged)))))))
      (pcase-dolist (`(,pos ,drop ,link) plan)
        (org-with-point-at pos
          (cond (drop (delete-region (point) (line-beginning-position 2)))
                ((and link (looking-at eet--checkbox-re))
                 (replace-match link t t nil 2))))
        (set-marker pos nil))
      (when unlogged
        (message "Not logged, no recipe: %s" (string-join (nreverse unlogged) ", ")))
      (nreverse cooked)))

  (defun eet--log (id note time)
    "Add NOTE, stamped TIME, to the LOGBOOK of recipe ID in the current buffer."
    (org-with-wide-buffer
     (when-let* ((pos (org-find-property "ID" id)))
       (goto-char pos)
       (let ((org-log-into-drawer "LOGBOOK"))
         (goto-char (org-log-beginning t)))
       (insert (format "- Note taken on %s \\\\\n  %s\n"
                       (format-time-string (org-time-stamp-format t t) time)
                       note)))))

  (defun eet--log-cooked (cooked time)
    "Write COOKED, (ID . NOTE) pairs, into recepten.org, stamped TIME."
    (when-let* ((cooked)
                (buf (eet--recipes-buffer)))
      (with-current-buffer buf
        (let ((save (not (buffer-modified-p))))
          (pcase-dolist (`(,id . ,note) cooked)
            (eet--log id note time))
          (if save
              (save-buffer)
            (message "%s had unsaved changes; save it to share the cooking log"
                     (buffer-name)))))))
