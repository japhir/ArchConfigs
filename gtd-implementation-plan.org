** project audit and keyword-less children
=my/gtd-project-audit= reports live projects with no action at all (EMPTY) or
more than one child =NEXT= (N-NEXT: demote the extras to =TODO=). The project
heading's own keyword does not count, so a checklist-model project (heading
=NEXT= + checkboxes) is neither.

=my/gtd-keywordless-project-children= lists sub-headings under a live project
that carry no keyword: candidate =TODO= (ADR 0001), or notes. With a prefix
argument (=C-u=) it walks them instead, shows each in place and asks whether
to make it =TODO=; saying yes goes through =org-todo=, so =CREATED= is stamped
by the usual hook. Answer =n= for notes; nothing is changed without a yes.
#+begin_src emacs-lisp
  (defun my/gtd--live-project-p ()
    "Non-nil when the heading at point is a :prj: (local tag) that is not done."
    (and (member "prj" (org-get-tags nil t))
         (not (org-entry-is-done-p))))

  (defun my/gtd-project-audit ()
    "Report live :prj: headings that are EMPTY (no action) or have >1 NEXT.
  The project heading's own keyword does not count: a heading carrying NEXT
  with a checklist body is the checklist model, not a 2-NEXT project."
    (interactive)
    (require 'cl-lib)
    (let (report)
      (org-map-entries
       (lambda ()
         (when (my/gtd--live-project-p)
           (let ((heading (org-get-heading t t t t))
                 (file (file-name-nondirectory (buffer-file-name)))
                 (top (point))
                 (next 0) (keyworded 0) (checks 0))
             (save-restriction
               (org-narrow-to-subtree)
               (org-map-entries
                (lambda ()
                  (unless (= (point) top)
                    (when (org-get-todo-state) (cl-incf keyworded))
                    (when (equal (org-get-todo-state) "NEXT") (cl-incf next))))
                t 'tree)
               (goto-char (point-min))
               (while (re-search-forward "^[ \t]*[-+*] \\[[ X-]\\]" nil t)
                 (cl-incf checks)))
             (let (flags)
               (when (and (zerop keyworded) (zerop checks)) (push "EMPTY" flags))
               (when (> next 1) (push (format "%d-NEXT" next) flags))
               (when flags
                 (push (format "%-12s %s  [%s]" file heading (string-join flags ","))
                       report))))))
       nil 'agenda 'archive 'comment)
      (with-current-buffer (get-buffer-create "*GTD project audit*")
        (erase-buffer)
        (insert "Live projects needing attention (EMPTY = no action at all; N-NEXT = demote extras to TODO)\n\n")
        (insert (string-join (nreverse report) "\n"))
        (goto-char (point-min))
        (display-buffer (current-buffer)))))

  (defun my/gtd-keywordless-project-children (&optional visit)
    "Sub-headings under a live :prj: that carry no todo keyword.
  Without VISIT, list them in a buffer. With VISIT (interactively: a prefix
  argument), walk them instead: show each and ask whether to make it TODO,
  which also stamps CREATED through the todo-state hook."
    (interactive "P")
    (let (report)
      (org-map-entries
       (lambda ()
         (when (my/gtd--live-project-p)
           (let ((proj (org-get-heading t t t t))
                 (top (point)))
             (save-restriction
               (org-narrow-to-subtree)
               (org-map-entries
                (lambda ()
                  (unless (or (= (point) top) (org-get-todo-state))
                    (push (list proj (org-get-heading t t t t) (point-marker)) report)))
                t 'tree)))))
       nil 'agenda 'archive 'comment)
      (setq report (nreverse report))
      (if visit
          (dolist (item report)
            (pcase-let ((`(,proj ,child ,m) item))
              (switch-to-buffer (marker-buffer m))
              (goto-char m)
              (org-fold-show-context)
              (when (y-or-n-p (format "%s -> %s: make it TODO? " proj child))
                (org-todo "TODO"))))
        (with-current-buffer (get-buffer-create "*GTD keywordless children*")
          (erase-buffer)
          (insert "Project sub-headings with no keyword (decide: TODO? note? promote NEXT?)\n"
                  "C-u M-x my/gtd-keywordless-project-children walks them and asks.\n\n")
          (dolist (item report)
            (insert (format "%s  ->  %s\n" (car item) (cadr item))))
          (goto-char (point-min))
          (display-buffer (current-buffer))))))
#+end_src
